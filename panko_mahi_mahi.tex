\section{Crispy Panko Mahi-Mahi with Warm French Lentil and Roasted Carrot Salad}
\begin{recipe}

\pre{
    Serves 5. About 45 minutes. Roast the carrots at 425\degree{}. Cook lentils and mix the vinaigrette up to 2 days ahead; refrigerate separately. For kids, keep some lentils and carrots undressed and serve the yogurt sauce as a dip.
}

\subsection{For the fish}
\ingredients{
    1\sfrac{3}{4}--2 & lb mahi-mahi, cut into 5 portions \\
    1 & cup panko breadcrumbs \\
    \sfrac{1}{3} & cup finely grated Parmesan \\
    2 & tablespoons olive oil \\
    2 & tablespoons mayonnaise \\
    1 & tablespoon Dijon mustard \\
    & zest of 1 lemon \\
    \sfrac{1}{2} & teaspoon garlic powder \\
    \sfrac{1}{2} & teaspoon kosher salt \\
    & pepper \\
}

\subsection{For the lentil-carrot salad}
\ingredients{
    1\sfrac{1}{4} & cups dry French green lentils \\
    1\sfrac{1}{2} & lb carrots, cut into \sfrac{1}{2}-inch pieces \\
    2 & tablespoons olive oil, for roasting \\
    1 & small shallot, finely minced \\
    \sfrac{1}{3} & cup fresh parsley, chopped \\
    2 & tablespoons lemon juice \\
    1 & tablespoon red wine vinegar \\
    1 & tablespoon extra-virgin olive oil \\
    2 & teaspoons Dijon mustard \\
    2 & teaspoons honey \\
    & salt and black pepper \\
}

\subsection{For the lemon-yogurt sauce}
\ingredients{
    \sfrac{3}{4} & cup plain Greek yogurt \\
    1 & tablespoon lemon juice \\
    1 & teaspoon honey \\
    1 & small garlic clove, grated (optional) \\
    1--2 & tablespoons water, to thin \\
    & salt \\
}

\subsection{Roast the carrots}
Heat the oven to 425\degree{}. Toss the carrots with 2 tablespoons olive oil, salt, and pepper. Spread on a sheet pan and roast for 25--30 minutes, turning once, until browned and tender.

\subsection{Simmer the lentils}
Rinse the lentils. Add them to a saucepan with 4 cups water and \sfrac{1}{2} teaspoon salt. Bring to a boil, then gently simmer for 20--30 minutes, until just tender. Drain well.

\subsection{Coat the fish}
Mix the panko, Parmesan, 2 tablespoons olive oil, lemon zest, garlic powder, salt, and pepper. Stir the mayonnaise and Dijon together. Pat the fish dry, spread the mayonnaise-Dijon mixture on top, and press on the crumbs. Set on a lightly oiled second pan.

\subsection{Bake the mahi-mahi}
When the carrots have about 15 minutes left, add the fish to the oven. Bake for 12--16 minutes (start checking early for thin pieces), until the thickest part reaches 145\degree{}. Optionally broil for 30--60 seconds to brown the crust.

\subsection{Dress the warm salad}
Whisk together the lemon juice, vinegar, extra-virgin olive oil, 2 teaspoons Dijon, and 2 teaspoons honey. Toss with the warm lentils, roasted carrots, shallot, and parsley. Season to taste.

\subsection{Sauce and serve}
Mix the yogurt, lemon juice, honey, optional garlic, salt, and enough water to make the sauce spoonable. Serve the fish with the warm salad and sauce on the side.

\tip{Cooking order: 0--10 minutes, prep and roast carrots; 10--15, start lentils; 15--25, coat fish and mix sauces; 25--30, add fish to oven; 30--40, drain lentils and finish carrots; 40--45, toss and serve.}

\end{recipe}
