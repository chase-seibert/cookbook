\section{Crispy Panko Mahi-Mahi with Warm French Lentil and Roasted Carrot Salad}
\begin{recipe}

\pre{
    Serves 5. About 45 minutes. Roast at 425\degree{}. Cook lentils and mix vinaigrette up to 2 days ahead; refrigerate separately. For kids, serve some lentils and carrots undressed, with yogurt sauce for dipping.
}

\subsection{Fish}
\ingredients{
    1\sfrac{3}{4}--2 & lb mahi-mahi, in 5 portions \\
    1 & cup panko \\
    \sfrac{1}{3} & cup finely grated Parmesan \\
    2 & tablespoons olive oil \\
    2 & tablespoons mayonnaise \\
    1 & tablespoon Dijon mustard \\
    & zest of 1 lemon \\
    \sfrac{1}{2} & teaspoon garlic powder \\
    \sfrac{1}{2} & teaspoon kosher salt \\
    & black pepper \\
}

\subsection{Lentil-carrot salad}
\ingredients{
    1\sfrac{1}{4} & cups French green lentils \\
    1\sfrac{1}{2} & lb carrots, in \sfrac{1}{2}-inch pieces \\
    2 & tablespoons olive oil, for roasting \\
    1 & small shallot, minced \\
    \sfrac{1}{3} & cup chopped parsley \\
    2 & tablespoons lemon juice \\
    1 & tablespoon red wine vinegar \\
    1 & tablespoon extra-virgin olive oil \\
    2 & teaspoons Dijon mustard \\
    2 & teaspoons honey \\
    & salt and black pepper \\
}

\subsection{Yogurt sauce}
\ingredients{
    \sfrac{3}{4} & cup plain Greek yogurt \\
    1 & tablespoon lemon juice \\
    1 & teaspoon honey \\
    1 & small garlic clove, grated (optional) \\
    1--2 & tablespoons water \\
    & salt \\
}

\subsection{Roast and simmer}
Heat oven to 425\degree{}. Toss carrots with 2 tablespoons oil, salt, and pepper; roast on a sheet pan 25--30 minutes, turning once. Rinse lentils; simmer in 4 cups water with \sfrac{1}{2} teaspoon salt for 20--30 minutes, until just tender. Drain well.

\subsection{Coat and bake fish}
Mix panko, Parmesan, 2 tablespoons oil, lemon zest, garlic powder, salt, and pepper. Stir mayonnaise with Dijon. Pat fish dry, spread mixture on top, and press on crumbs. Put on a lightly oiled pan. When carrots have about 15 minutes left, bake fish 12--16 minutes, until the thickest part reaches 145\degree{} (check thin pieces early). Broil 30--60 seconds to brown, if desired.

\subsection{Finish salad and sauce}
Whisk lemon juice, vinegar, extra-virgin olive oil, Dijon, and honey. Toss with warm lentils, roasted carrots, shallot, and parsley; season. Thin yogurt with lemon juice, honey, optional garlic, salt, and enough water to spoon. Serve with fish.

\endgroup
\end{recipe}
